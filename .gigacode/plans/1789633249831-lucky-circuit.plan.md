# Chaos Test for HurriCache Java Client

## Цель
Многопотестный тест, который создаёт хаотичную нагрузку на кластер, создавая и модифицируя различные типы контейнеров и атомиков параллельно, считая ошибки (таймауты, PERMISSION_DENIED и др.).

## Архитектура

### Фаза префила

Создаёт 50% ключей каждого типа ДО начала хаос-теста:

```java
private void runPrefillPhase(ChaosTestConfig config, ChaosTestStats stats) {
    int prefillKeys = (int) (config.totalKeys * config.prefillRatio / 100.0); // 25 000
    int keysPerType = prefillKeys / PRELOAD_TYPES.length; // ~2778 на тип
    
    var executor = Executors.newVirtualThreadPerTaskExecutor();
    List<CompletableFuture<?>> futures = new ArrayList<>();
    
    for (int t = 0; t < config.numThreads; t++) {
        futures.add(CompletableFuture.runAsync(() -> {
            var random = ThreadLocalRandom.current();
            for (int i = 0; i < keysPerType; i++) {
                String key = generateKeyForType(random, PRELOAD_TYPES[t % PRELOAD_TYPES.length]);
                try {
                    preloadOperation(key, t, config, stats);
                    stats.prefillCreated.incrementAndGet();
                } catch (Exception e) {
                    stats.prefillFailed.incrementAndGet();
                }
            }
        }, executor));
    }
    
    CompletableFuture.allOf(futures.toArray(new CompletableFuture[0])).join();
}

private void preloadOperation(String key, int clientId, ChaosTestConfig config, ChaosTestStats stats) {
    switch (preloadType) {
        case SCALAR -> client.createKeyValue(key, "value".getBytes()).join();
        case VECTOR -> client.createVector(key, generatePayloads(config.elementsPerContainer)).join();
        case LIST -> client.createList(key, generatePayloads(config.elementsPerContainer)).join();
        case QUEUE -> client.createQueue(key, generatePayloads(config.elementsPerContainer)).join();
        case SET -> client.createSet(key, generatePayloads(config.elementsPerContainer)).join();
        case MAP -> client.createMap(key, generateMapEntries(config.elementsPerContainer)).join();
        case ORDERED_MAP -> client.createOrderedMap(key, generateOrderedMapEntries(config.elementsPerContainer)).join();
        case ORDERED_SET -> client.createOrderedSet(key, generateOrderedPayloads(config.elementsPerContainer)).join();
        case ATOMIC -> client.atomicCreate(key, ThreadLocalRandom.current().nextLong(1, 1_000_000)).join();
    }
}
```

### Логика теста

```java
@ExtendWith(JupiterExtensionRegistration.class)
public class ChaosClusterTest extends TestBaseCluster {
    
    private static final String[] PRELOAD_TYPES = {
        "scalar", "vector", "list", "queue", "set", "map", "omap", "oset", "atomic"
    };
    
    @Test
    @DisplayName("Chaos test: concurrent operations on all container types with prefill")
    void testChaosCluster() throws Exception {
        ChaosTestConfig config = new ChaosTestConfig();
        // По умолчанию: 30 минут, 2 потока, 100мс интервал
        config.totalKeys = 50_000;
        config.elementsPerContainer = 100;
        config.enablePrefill = true;
        config.prefillRatio = 50;
        
        ChaosTestStats stats = new ChaosTestStats();
        
        // Фаза 1: Префил — создаём 50% ключей
        if (config.enablePrefill) {
            System.out.println("[CHaos] Phase 1: Prefill " + config.prefillRatio + "% of " + config.totalKeys + " keys...");
            runPrefillPhase(config, stats);
            System.out.println("[CHaos] Prefill complete: " + stats.prefillCreated.get() + " created, " + 
                             stats.prefillFailed.get() + " failed");
        }
        
        // Фаза 2: Хаос — параллельные операции
        System.out.println("[CHaos] Phase 2: Chaos for " + (config.durationMs / 60_000) + " minutes with " + config.numThreads + " threads...");
        runChaosPhase(config, stats);
        
        // Выводим статистику
        printStats(stats);
        
        // Проверяем, что процент успешных операций > 90%
        double successRate = (double) stats.successfulOperations.get() / stats.totalOperations.get();
        assertTrue(successRate > 0.9, 
            "Success rate " + (successRate * 100) + "% is below 90% threshold");
    }
    
    private void runChaosPhase(ChaosTestConfig config, ChaosTestStats stats) {
        var executor = Executors.newVirtualThreadPerTaskExecutor();
        List<CompletableFuture<?>> futures = new ArrayList<>();
        
        long endTime = System.currentTimeMillis() + config.durationMs;
        
        for (int t = 0; t < config.numThreads; t++) {
            futures.add(CompletableFuture.runAsync(() -> {
                runChaosWorker(t, config, stats, endTime);
            }, executor));
        }
        
        CompletableFuture.allOf(futures.toArray(new CompletableFuture[0])).join();
    }
    
    private void runChaosWorker(int threadId, ChaosTestConfig config, ChaosTestStats stats, long endTime) {
        var random = ThreadLocalRandom.current();
        
        while (System.currentTimeMillis() < endTime) {
            // Выбираем случайный ключ (из существующих или новый)
            String key = random.nextBoolean()
                ? generateExistingKey(random, stats)
                : generateRandomKey(random);
            
            ChaosOperation op = randomOperation(random);
            
            long startTime = System.currentTimeMillis();
            try {
                op.execute(key, threadId, stats);
                stats.successfulOperations.incrementAndGet();
            } catch (Exception e) {
                classifyError(e, stats);
            } finally {
                stats.totalOperations.incrementAndGet();
                long elapsed = System.currentTimeMillis() - startTime;
                stats.totalDurationMs.addAndGet(elapsed);
            }
            
            if (config.operationIntervalMs > 0) {
                Thread.sleep(config.operationIntervalMs);
            }
        }
    }
}
```

```java
private String generateRandomKey(ThreadLocalRandom random) {
    String[] prefixes = {"scalar-", "vector-", "list-", "queue-", "set-", "map-", "omap-", "oset-", "atomic-"};
    String prefix = prefixes[random.nextInt(prefixes.length)];
    return prefix + random.nextInt(1, 1_000_001);
}

private String generateKeyForType(ThreadLocalRandom random, String type) {
    return type + "-" + random.nextInt(1, 1_000_001);
}

private ChaosOperation randomOperation(ThreadLocalRandom random) {

private List<Payload> generatePayloads(int count) {
    var random = ThreadLocalRandom.current();
    var payloads = new ArrayList<Payload>(count);
    for (int i = 0; i < count; i++) {
        payloads.add(Payload.of(("elem_" + random.nextInt(10000)).getBytes()));
```

## Вывод статистики

```java
private void printStats(ChaosTestStats stats) {
    System.out.println("\n========== Chaos Test Results ==========");
    System.out.println("Total operations: " + stats.totalOperations.get());
    System.out.println("Successful: " + stats.successfulOperations.get());
    System.out.println("Success rate: " + ((double) stats.successfulOperations.get() / stats.totalOperations.get() * 100) + "%");
    System.out.println("Timeout errors: " + stats.timeoutErrors.get());
    System.out.println("Permission denied: " + stats.permissionDeniedErrors.get());
    System.out.println("Cant lock: " + stats.cantLockErrors.get());
    System.out.println("Cant unlock: " + stats.cantUnlockErrors.get());
    System.out.println("Not found: " + stats.notFoundErrors.get());
    System.out.println("Other errors: " + stats.otherErrors.get());
    System.out.println("Avg operation time: " + (stats.totalDurationMs.get() / (double) stats.totalOperations.get()) + "ms");
    if (stats.prefillCreated.get() > 0) {
        System.out.println("\n--- Prefill Stats ---");
        System.out.println("Created: " + stats.prefillCreated.get());
        System.out.println("Failed: " + stats.prefillFailed.get());
    }
    System.out.println("========================================\n");
}
```
